+## Composer typing under load
+
+The canonical browser diagnostic is `webviews/bench/composer-typing.bench.ts`:
+
+```sh
+cmux-ci test --setup playwright --timeout 2400 -- bash -lc 'cd webviews && bun install --frozen-lockfile && bun bench/composer-typing.bench.ts --mode L0,L1,L2,L3 --engines chromium,webkit --lengths 60 --keys 500 --gap 20'
+```
+
+`--modes idle,stream` retains its Chromium render-cost comparison. `--mode L0,L1,L2,L3`
+adds idle, focused reply/tool stream (60 chunks and 2,000 tool lines per second), ten
+other streaming documents (five visible, five hidden), and a focused 100k-row stream.
+`--stream` and `--tool-lines` change the focused load. Every load ends explicitly or
+at its 120-second safety deadline. L2 is an iframe approximation, not native tab or
+process placement; L4/L5 and daemon wakeups require the native Mac harness.
+
+`COMPOSER-TYPING` rows include exact ordered key/text integrity, load continuity/rates,
+p50/p95/p99/max, frame bursts, frame intervals over 16 ms, and raw samples. Invalid
+load/cadence receipts never meet the target. Budgets are report-only (16 ms p95 and
+33 ms p99, at least 200 keys); the browser clock starts at DOM keydown, so these
+rows cannot certify native admission or physical presentation. Long Tasks observes
+only tasks of at least 50 ms: zero means none observed above that threshold, and
+unsupported engines (including WebKit) report null, never a fabricated zero.
+`valid` establishes experiment integrity, not a latency pass. Short diagnostic runs remain
+valid but are not `acceptance_eligible`; acceptance requires `target_met` (valid load,
+at least 200 keys, and both latency limits). Consumers must never interpret `valid` alone
+as a budget pass.
+Tool output updates carry the accumulated output at each reply tick, matching the mock
+tool-update format. This deliberately stresses growing frontend payloads; it does not
+model native provider transport. Cross-document epoch receipts require wall-clock
+agreement within 2 ms or invalidate the run.

+import { MockAcpmuxSocket } from "./mock";

+// Benchmark-only traffic, through the mock daemon and the real reducer. No frame observer,
+// scrolling or layout reads: those would add measurement work to the key's critical path.
+export type TypingLoadOptions = {
+  stop?: boolean;
+  chunks_per_second?: number;
+  tool_lines_per_second?: number;
+  seconds?: number;
+};
+let active:
+  | {
+      stopped: boolean;
+      done: Promise<unknown>;
+      receipt: {
+        running: boolean;
+        started_epoch_ms: number;
+        finished_epoch_ms: number | null;
+        deltas: number;
+        toolLines: number;
+        elapsedMs: number;
+        maxGapMs: number;
+        maxScheduleOffsetMs: number;
+        error?: string;
+      };
+    }
+  | undefined;
+
+function snapshot(receipt: Record<string, unknown>) {
+  return { ...receipt, epoch_clock_error_ms: Date.now() - (performance.timeOrigin + performance.now()) };
+}
+
+export async function typingLoad(
+  session: string | undefined,
+  options: TypingLoadOptions = {},
+): Promise<Record<string, unknown>> {
+  if (options.stop) {
+    if (!active) return { error: "no typing load" };
+    const current = active;
+    current.stopped = true;
+    await current.done;
+    return snapshot(current.receipt);
+  }
+  // With no rate, read the receipt without starting or resetting the generator.
+  if (options.chunks_per_second === undefined) return active ? snapshot(active.receipt) : { error: "no typing load" };
+  if (active?.receipt.running) return { error: "typing load already running" };
+  const mock = MockAcpmuxSocket.live;
+  if (!mock || !session) return { error: "typing load requires a seeded mock session" };
+  const rate = options.chunks_per_second;
+  const lines = options.tool_lines_per_second ?? 0;
+  const seconds = options.seconds ?? 120;
+  if (
+    ![rate, lines, seconds].every(Number.isFinite) ||
+    rate <= 0 ||
+    rate > 1000 ||
+    lines < 0 ||
+    lines > 10000 ||
+    seconds <= 0 ||
+    seconds > 120
+  )
+    return { error: "typing load requires rate 1..1000, lines 0..10000, duration (0,120]" };
+  const state = {
+    stopped: false,
+    done: Promise.resolve<unknown>(undefined),
+    receipt: {
+      running: true,
+      started_epoch_ms: performance.timeOrigin + performance.now(),
+      finished_epoch_ms: null as number | null,
+      deltas: 0,
+      toolLines: 0,
+      elapsedMs: 0,
+      maxGapMs: 0,
+      maxScheduleOffsetMs: 0,
+      error: undefined as string | undefined,
+    },
+  };
+  active = state;
+  state.done = mock
+    .streamLoadTurn(session, {
+      seconds,
+      chunkChars: 32,
+      chunkMs: 1000 / rate,
+      toolEvery: 0,
+      fixedSchedule: true,
+      toolLinesPerSecond: lines,
+      stopped: () => state.stopped,
+      progress: (counts) => Object.assign(state.receipt, counts),
+    })
+    .catch((error: unknown) => {
+      state.receipt.error = String(error);
+    })
+    .finally(() => {
+      state.receipt.running = false;
+      state.receipt.finished_epoch_ms = performance.timeOrigin + performance.now();
+    });
