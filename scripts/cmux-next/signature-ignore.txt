# Stack frames that crash_signature.py drops before it takes the top 3 frames.
#
# The list matches ClusterFuzz's STACK_FRAME_IGNORE_REGEXES
# (google/clusterfuzz src/clusterfuzz/stacktraces/constants.py): frames of the
# runtime, libc, allocators, assertion helpers and sanitizers say how a process
# died, not where the bug is. One Python regex per line, searched in the
# normalized function name. A line that starts with `image:` is searched in the
# frame's image (binary) name instead, and one that starts with `file:` in the
# frame's source file (Rust backtraces print it on an `at` line). `#` starts a
# comment.

# --- Images: the whole binary is runtime or system plumbing.
image:^libsystem_
image:^libdyld
image:^dyld$
image:^libdispatch
image:^libobjc
image:^libswift
image:^libc\+\+
image:^libclang_rt\.
image:^libc\.so
image:^libpthread
image:^ld-linux
image:^libgcc_s

# --- Source files: Rust std, core and alloc (their inlined helpers print as
# bare `index`, `unwrap`, `expect`), crates from the registry, and the panic hook.
file:^/rustc/
file:/\.cargo/(registry|git)/
file:(^|/)cmux-panic-log/
