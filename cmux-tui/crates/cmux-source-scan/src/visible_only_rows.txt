The rows of the visible-only check (src/visible_only.rs), read by visible_only_tests.rs. Each row
starts with `=== KIND ARGS :: TITLE`; its source (a file under a `ui` module) runs to the next
`===` line. KIND is
  passes                        the check finds nothing
  flags LINE KIND [& LINE KIND] the check finds exactly these (KIND walk or clone), each once
The first rows are the shapes of the candidates read in acpmux and cmux-tui at the check's start.

=== passes :: bounded walks: a range, a range slice, skip/take, a constant, text, an array
fn draw_rows(frame: &mut ratatui::Frame, rows: &[Row], area: Rect) {
    for x in area.x..area.x + area.width {
        frame.paint(x);
    }
    for row in rows[offset..offset + h].iter() {
        frame.draw(row);
    }
    for row in rows.iter().skip(offset).take(h) {
        frame.draw(row);
    }
    let first = rows.iter().next();
    for (k, v) in KEYS.iter() {
        frame.label(k, v);
    }
    let w: usize = label.chars().map(|c| c.len_utf8()).sum();
    for word in label.split_whitespace() {
        frame.word(word);
    }
    for part in [a, b, c] {
        frame.part(part);
    }
    let shown = rows[a..b].to_vec();
}

fn draw_tail(frame: &mut ratatui::Frame, rows: &[Row]) {
    let newest = rows.iter().last();
    let at = rows.iter().nth(selected);
    let every_other: Vec<_> = rows.iter().step_by(2).collect();
    let head: Vec<_> = rows.iter().take_while(|r| r.pinned).collect();
    let shown: Vec<_> = rows.iter().filter(|r| r.visible).take(h).collect();
    let first_unread = rows.iter().filter(|r| r.unread).next();
}
=== passes :: a bound after adapters that read one item per item still bounds the walk
fn draw_numbered(frame: &mut ratatui::Frame, rows: &[Row]) {
    for (i, row) in rows.iter().enumerate().skip(offset).take(h) {
        frame.draw(i, row);
    }
    let tail: Vec<_> = rows.iter().rev().map(|r| r.id).take(h).collect();
}
